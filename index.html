async function load() { try { const res = await fetch("data.json"); const data = await res.json(); const list = data.data || []; list.sort((a, b) => (b.wagered || 0) - (a.wagered || 0)); let html = ""; list.slice(0, 10).forEach((u, i) => { const medal = i === 0 ? "🥇" : i === 1 ? "🥈" : i === 2 ? "🥉" : `#${i + 1}`; html += `
${medal} ${u.username || "Unknown"}
$${Number(u.wagered || 0).toLocaleString()}
`; }); document.getElementById("list").innerHTML = html; } catch (e) { document.getElementById("list").innerHTML = "Error loading data"; } }
